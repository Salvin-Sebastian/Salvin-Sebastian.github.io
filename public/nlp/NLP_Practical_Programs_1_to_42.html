import nltk

sent = "The student solved the puzzle"
tags = nltk.pos_tag(nltk.word_tokenize(sent))

# Rules: words before verb = Subject, verb = Predicate, words after = Object
subject, verb, obj = [], [], []
state = 'SUBJ'
for w, t in tags:
    if t.startswith('VB'):
        state = 'VERB'
        verb.append(w)
    elif state == 'VERB':
        state = 'OBJ'
        obj.append(w)
    elif state == 'SUBJ':
        subject.append(w)
    else:
        obj.append(w)

print("Subject:  ", " ".join(subject))
print("Predicate:", " ".join(verb))
print("Object:   ", " ".join(obj))

def fsa_recognize_ing(word):
    # Recognizes words ending with 'ing'
    state = 0
    for ch in word.lower():
        if state == 0: state = 1 if ch == 'i' else 0
        elif state == 1: state = 2 if ch == 'n' else (1 if ch == 'i' else 0)
        elif state == 2: state = 3 if ch == 'g' else (1 if ch == 'i' else 0)
        elif state == 3: state = 1 if ch == 'i' else 0
    return state == 3

for w in ["sing", "playing", "walk"]:
    print(f"'{w}': {fsa_recognize_ing(w)}")

def morphological_analyze(word):
    prefixes = ['un', 're', 'dis']
    suffixes = ['ness', 'ing', 'ed', 'ly']
    stem = word.lower()

    prefix = next((p for p in prefixes if stem.startswith(p)), "")
    if prefix: stem = stem[len(prefix):]

    suffix = next((s for s in suffixes if stem.endswith(s)), "")
    if suffix: stem = stem[:-len(suffix)]

    return prefix, stem, suffix

print("unhappiness ->", morphological_analyze("unhappiness"))
print("replaying   ->", morphological_analyze("replaying"))

PROGRAM 17 17_hidden_markov_model_sequence_prediction.py
Sequence Models, POS Tagging & Classification
Problem Statement:

Implement a simple Hidden Markov Model (HMM) for sequence prediction

Python 3 Source Code 
states = ['Sunny', 'Rainy']
start_p = {'Sunny': 0.6, 'Rainy': 0.4}
trans_p = {'Sunny': {'Sunny': 0.7, 'Rainy': 0.3}, 'Rainy': {'Sunny': 0.4, 'Rainy': 0.6}}
emit_p  = {'Sunny': {'walk': 0.6, 'shop': 0.4}, 'Rainy': {'walk': 0.1, 'shop': 0.9}}

obs = ['walk', 'shop']
# Viterbi decoding algorithm
v0 = {s: start_p[s] * emit_p[s][obs[0]] for s in states}
v1 = {s: max(v0[prev] * trans_p[prev][s] * emit_p[s][obs[1]] for prev in states) for s in states}
best_state = max(v1, key=v1.get)

print("Observations:      ", obs)
print("Likely Final State:", best_state)
Expected Terminal Output

sent = "they will fish and opened the can".split()
# Step 1: Initial baseline tagger (all Nouns)
tags = ['NN'] * len(sent)

# Step 2: Contextual transformation rules
for i in range(len(sent)):
    if i > 0 and sent[i-1] in ['will', 'can']: tags[i] = 'VB'  # Modal -> Verb
    elif sent[i].endswith('ed'): tags[i] = 'VBD'              # Suffix -ed -> Past Verb

lexicon = {'the': 'DT', 'is': 'VBZ', 'fast': 'JJ'}

def tag_unknown(w):
    if w.lower() in lexicon: return lexicon[w.lower()]
    if w.isdigit(): return 'CD'        # Number
    if w.endswith('ing'): return 'VBG' # Gerund
    if w.endswith('ed'): return 'VBD'  # Past verb
    if w.endswith('ly'): return 'RB'   # Adverb
    return 'NN'                        # Default open-class noun

# Sequence-to-sequence translation of words and phrases (English -> French)

seq_model = {
    "hello": "bonjour",
    "how are you": "comment vas-tu",
    "thank you": "merci",
    "good morning": "bonjour"
}

def translate_sequence(sentence):
    text = sentence.lower().strip()
    if text in seq_model:
        return seq_model[text]
    word_map = {"how": "comment", "are": "vas", "you": "tu", "thank": "merci"}
    return " ".join(word_map.get(w, w) for w in text.split())

for s in ["Hello", "How are you", "Thank you"]:
    print(f"'{s}' -> '{translate_sequence(s)}'")

Generate text summaries using extractive methods

from collections import Counter
from nltk.tokenize import sent_tokenize, word_tokenize

text = "NLP is a branch of AI. It helps computers understand human language. NLP is useful for translation and chatbots."
sents = sent_tokenize(text)
words = [w.lower() for w in word_tokenize(text) if w.isalnum()]
freq = Counter(words)

# Score sentences by word frequency
scores = {s: sum(freq[w.lower()] for w in word_tokenize(s) if w.isalnum()) for s in sents}
top_sent = max(scores, key=scores.get)

print("Original Sentences:", len(sents))
print("Extractive Summary:", top_sent)
